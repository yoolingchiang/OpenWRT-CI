#无 WiFi 的 IPQ60XX 设备配置（当前：兆能 ZN-M2，512MB + 128MB NAND）
#本配置不绑定源码，ImmortalWrt 与 LibWrt 两条线共用同一份。
#设备专属取舍请改 Config/IPQ60XX-ZNM2-Override.txt（按设备关键词自动匹配）。

#设备平台
CONFIG_TARGET_qualcommax=y
CONFIG_TARGET_qualcommax_ipq60xx=y
#设备列表：兆能 ZN-M2（NAND，无 WiFi）
CONFIG_TARGET_DEVICE_qualcommax_ipq60xx_DEVICE_zn_m2=y

#无 WiFi：关闭 ath11k 驱动与固件
#（Settings.sh 检测到 kmod-ath11k=n 后会把 DTS 切成 ipq6018-nowifi.dtsi，
#  给 NSS 少预留内存，512MB 机型很关键；固件文件名也会带上 wifi-no 标记）
CONFIG_PACKAGE_kmod-ath=n
CONFIG_PACKAGE_kmod-ath11k=n
CONFIG_PACKAGE_kmod-ath11k-ahb=n
CONFIG_PACKAGE_kmod-ath11k-pci=n
CONFIG_PACKAGE_ath11k-firmware-ipq6018=n
CONFIG_PACKAGE_ath11k-firmware-ipq6018-ddwrt=n
CONFIG_PACKAGE_ath11k-firmware-qcn9074=n
CONFIG_PACKAGE_ath11k-firmware-qcn9074-ddwrt=n

#插件清单（128MB NAND 空间紧张，比带 WiFi 那份克制得多）
#科学上网：只用 nikki
CONFIG_PACKAGE_luci-app-nikki=y
CONFIG_PACKAGE_luci-i18n-nikki-zh-cn=y
#常用工具
CONFIG_PACKAGE_luci-app-ddns-go=y
CONFIG_PACKAGE_luci-app-netwizard=y
CONFIG_PACKAGE_luci-app-easytier=y
